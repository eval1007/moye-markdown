async function loadDashboard({ fetchSales, fetchAds, fetchStock }) {
  const [sales, ads, stock] = await Promise.allSettled([
    fetchSales(),
    fetchAds(),
    fetchStock(),
  ]);

  if (sales.status === "rejected") {
    return { status: 503, body: { code: "SALES_UNAVAILABLE" } };
  }

  const body = { sales: sales.value, missingModules: [] };
  for (const [name, result] of [["ads", ads], ["stock", stock]]) {
    if (result.status === "fulfilled") {
      body[name] = result.value;
    } else {
      body.missingModules.push(name);
    }
  }

  return { status: 200, body };
}

SELECT
    snapshot_id,
    payload,
    refreshed_at,
    CASE
        WHEN fresh_until > :request_time THEN 'fresh'
        ELSE 'stale'
    END AS freshness
FROM dashboard_snapshots
WHERE store_id = :store_id
  AND report_date = :report_date
  AND usable_until > :request_time
ORDER BY refreshed_at DESC, snapshot_id DESC
LIMIT 1;
